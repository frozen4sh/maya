def mirrorPoints(*args):
    try:
        sel = cmds.ls(sl=1)
        cmds.select(topoSetup)
        loadSetup()
        cmds.warning('Done')
        cmds.select(sel)
        if len(sel) > 0:
            pntsA = cmds.listRelatives(ga, c=1, pa=1)
            pntsB = cmds.listRelatives(gb, c=1, pa=1)
            lenPnt = len(pntsB)
            try:
                cmds.lockNode((pntsA[lenPnt]), lock=0)
                cmds.delete(pntsA[lenPnt])
                pntsA = cmds.listRelatives(ga, c=1, pa=1)
                pntsB = cmds.listRelatives(gb, c=1, pa=1)
            except:
                pass

            dupPntsA = []
            dupPntsB = []
            for sl in sel:
                for e, f in zip(pntsA, pntsB):
                    if sl == e or sl == f:
                        d = cmds.duplicate(e)[0]
                        dupPntsA.append(d)
                        d = cmds.duplicate(f)[0]
                        dupPntsB.append(d)

        else:
            pntsA = cmds.listRelatives(ga, c=1, pa=1)
            pntsB = cmds.listRelatives(gb, c=1, pa=1)
            if pntsB == None:
                cmds.warning('No Topo-Points found for Mirroring')
            else:
                lenPnt = len(pntsB)
                try:
                    cmds.lockNode((pntsA[lenPnt]), lock=0)
                    cmds.delete(pntsA[lenPnt])
                    pntsA = cmds.listRelatives(ga, c=1, pa=1)
                    pntsB = cmds.listRelatives(gb, c=1, pa=1)
                except:
                    pass

                pntsAB = (
                 pntsA, pntsB)
                for ee in pntsAB:
                    pv = [
                     'a']
                    ng = ['a']
                    for e in ee:
                        p = cmds.xform(e, t=1, ws=1, q=1)[0]
                        pe = [p, e]
                        if p < 0:
                            ng.append(pe)
                        else:
                            pv.append(pe)

                    xxx = [
                     [
                      len(pv), pv], [len(ng), ng]]
                    xxx.sort()
                    for e in xxx[0][1]:
                        try:
                            cmds.lockNode((e[1]), lock=0)
                            cmds.delete(e[1])
                        except:
                            pass

                pntsA = cmds.listRelatives(ga, c=1, pa=1)
                pntsB = cmds.listRelatives(gb, c=1, pa=1)
                dupPntsA = cmds.duplicate(pntsA)
                dupPntsB = cmds.duplicate(pntsB)
        if pntsA == None:
            cmds.warning('No Topo-Points found for Mirroring')
        else:
            for e, f in zip(dupPntsA, dupPntsB):
                unlockT(e)
                unlockT(f)
                vl = cmds.getAttr(e + '.translateX')
                cmds.setAttr(e + '.translateX', -vl)
                vl = cmds.getAttr(f + '.translateX')
                cmds.setAttr(f + '.translateX', -vl)
                lockT(e)
                lockT(f)
                cmds.lockNode(e)
                cmds.lockNode(f)

            pntsA = cmds.listRelatives(ga, c=1)
            pntsB = cmds.listRelatives(gb, c=1)
            ct = 1
            for e, f in zip(pntsA, pntsB):
                cmds.setAttr((e + '.text'), ct, type='string')
                cmds.setAttr((f + '.text'), ct, type='string')
                ct += 1

    except:
        cmds.warning('No Valid Retopo Setup or Points Found')

def snapRetopo(*args):
    try:
        cmds.select(topoSetup)
        loadSetup()
        cmds.warning('Retopo Mesh Creation in Progress')
        cmds.select(cl=1)
        try:
            pntsA = cmds.listRelatives(ga, c=1, pa=1)
            pntsB = cmds.listRelatives(gb, c=1, pa=1)
        except:
            cmds.warning('Please Create Marker Points')

        if pntsB == None:
            cmds.warning('Please Create Marker Points')
        else:
            lenPnt = len(pntsB)
            if lenPnt < 4:
                cmds.warning('Cannot create Retopo Mesh, Please create minimum 4 Marker Points or more')
            else:
                progres = cmds.window(t=' Creating Retopo Mesh ')
                cmds.columnLayout()
                progressControl = cmds.progressBar(width=380)
                cmds.showWindow(progres)
                progressInc = cmds.progressBar(progressControl, edit=True, pr=0)
                try:
                    cmds.lockNode((pntsA[lenPnt]), lock=0)
                    cmds.delete(pntsA[lenPnt])
                    pntsA = cmds.listRelatives(ga, c=1, pa=1)
                    pntsB = cmds.listRelatives(gb, c=1, pa=1)
                except:
                    pass

                jtsA = []
                pls = []
                for e in pntsA:
                    p = cmds.xform(e, t=1, ws=1, q=1)
                    cmds.select(cl=1)
                    jt = cmds.joint(radius=0.01, p=(p[0], p[1], p[2]))
                    cmds.select(cl=1)
                    pl = cmds.polyPlane(sx=1, sy=1, w=0.001, h=0.001)[0]
                    cmds.xform(pl, t=p, ws=1)
                    pls.append(pl)
                    jtsA.append(jt)

                progressInc = cmds.progressBar(progressControl, edit=True, pr=10)
                plsGrp = cmds.group(pls)
                ptc = cmds.polyUnite(ch=0, n='ptc')[0]
                lowMesh = cmds.duplicate(ma, n='lowMesh')[0]
                retopMesh = cmds.duplicate(ma, n='retopMesh_01')[0]
                cmds.polySmooth(lowMesh, ch=0, dv=1)
                progressInc = cmds.progressBar(progressControl, edit=True, pr=20)
                cmds.transferAttributes(ptc, lowMesh, transferPositions=1, transferNormals=0, transferUVs=0, transferColors=0, sampleSpace=0,
                  sourceUvSpace='map1',
                  targetUvSpace='map1',
                  searchMethod=3,
                  flipUVs=0,
                  colorBorders=1)
                progressInc = cmds.progressBar(progressControl, edit=True, pr=40)
                cmds.polyMergeVertex(lowMesh, d=0.005)
                progressInc = cmds.progressBar(progressControl, edit=True, pr=60)
                cmds.delete(lowMesh, ch=1)
                cmds.delete(ptc)
                cmds.skinCluster(jtsA, lowMesh)
                cmds.skinCluster(jtsA, retopMesh)
                cmds.copySkinWeights(lowMesh, retopMesh, noMirror=1, surfaceAssociation='closestPoint', influenceAssociation='closestJoint')
                for e, f in zip(jtsA, pntsB):
                    p = cmds.xform(f, t=1, ws=1, q=1)
                    cmds.xform(e, t=p, ws=1)

                cmds.select(retopMesh)
                cmds.CenterPivot()
                cmds.DeleteHistory()
                cmds.makeIdentity(apply=1, t=1, r=1, s=1, n=0, pn=1)
                progressInc = cmds.progressBar(progressControl, edit=True, pr=80)
                baseWrapMesh = cmds.duplicate(retopMesh, n='baseWrap')
                cmds.transferAttributes(mb, retopMesh, transferPositions=1, transferNormals=0, transferUVs=0, transferColors=0, sampleSpace=0,
                  sourceUvSpace='map1',
                  targetUvSpace='map1',
                  searchMethod=3,
                  flipUVs=0,
                  colorBorders=1)
                cmds.delete(retopMesh, ch=1)
                cmds.polyNormalPerVertex(retopMesh, ufn=1)
                cmds.polySoftEdge(retopMesh, a=180, ch=0)
                cmds.delete(retopMesh, ch=1)
                bs = cmds.blendShape(baseWrapMesh, retopMesh, n='shapeBlend', w=[(0, 1)])[0]
                vcnt = cmds.polyEvaluate(retopMesh, v=1)
                for e in range(vcnt):
                    cmds.setAttr(bs + '.inputTarget[' + str(0) + '].inputTargetGroup[0].targetWeights' + '[' + str(e) + ']', 0.0)

                try:
                    cmds.select('excludeSet')
                    vcnt = cmds.ls(sl=1, fl=1)
                    vtxs = []
                    for e in vcnt:
                        vtx = e.split('[')[0]
                        vtxN = vtx.split('.')[0]
                        if vtxN == ma:
                            vtxs.append(e)

                    for e in vtxs:
                        vtxNo = e.split('[')[1]
                        vtxNo = int(vtxNo.split(']')[0])
                        cmds.setAttr(bs + '.inputTarget[' + str(0) + '].inputTargetGroup[0].targetWeights' + '[' + str(vtxNo) + ']', 1)

                except:
                    pass

                cmds.delete(baseWrapMesh)
                cmds.setAttr(retopMesh + '.overrideEnabled', 0)
                cmds.select(retopMesh)
                cmds.hyperShade(assign='lambert1')
                cmds.parent(w=1)
                cmds.delete(jtsA, lowMesh)
                progressInc = cmds.progressBar(progressControl, edit=True, pr=100)
                cmds.deleteUI(progres)
    except:
        cmds.warning('No Valid Retopo Setup Found')
